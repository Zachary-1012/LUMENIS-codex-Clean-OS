# app/lib/features/ai01/ai01_intent.dart:1-66
1: /// Result of one AI-01 intent submission, parsed strictly from the real
2: /// `POST /api/experience/ai/intent` response.
3: ///
4: /// A 2xx response is never treated as success by itself: the server returns
5: /// `ok:true` with a non-`completed` execution status (for example `failed`
6: /// while the owned AI runtime is not bound), and `ok:false` with
7: /// `pending_approval`. Only an execution whose returned status is `completed`
8: /// is reported as completed, and Evidence is only "recorded" when the server
9: /// says so.
10: enum Ai01IntentKind { completed, notCompleted, pendingApproval }
11: 
12: class Ai01IntentResult {
13:   const Ai01IntentResult({
14:     required this.kind,
15:     required this.status,
16:     this.executionId,
17:     this.evidenceRecorded,
18:     this.auditId,
19:   });
20: 
21:   final Ai01IntentKind kind;
22: 
23:   /// Server-returned execution status (`completed`, `failed`, `blocked`,
24:   /// `rejected`, ...) or `pending_approval`.
25:   final String status;
26:   final String? executionId;
27: 
28:   /// `null` when the server did not report an execution (pending approval).
29:   final bool? evidenceRecorded;
30:   final String? auditId;
31: 
32:   factory Ai01IntentResult.fromResponse(Map<String, dynamic> response) {
33:     final status = response['status'];
34:     final auditId = response['audit_id'];
35:     if (response['ok'] == false &&
36:         status == 'pending_approval' &&
37:         response['requires_approval'] == true) {
38:       return Ai01IntentResult(
39:         kind: Ai01IntentKind.pendingApproval,
40:         status: 'pending_approval',
41:         auditId: auditId is String && auditId.isNotEmpty ? auditId : null,
42:       );
43:     }
44: 
45:     final executionId = response['execution_id'];
46:     final evidenceRecorded = response['evidence_recorded'];
47:     if (response['ok'] != true ||
48:         response['permission_status'] != 'granted' ||
49:         status is! String ||
50:         status.isEmpty ||
51:         executionId is! String ||
52:         executionId.isEmpty ||
53:         evidenceRecorded is! bool) {
54:       throw const FormatException('AI-01 intent response is not valid.');
55:     }
56:     return Ai01IntentResult(
57:       kind: status == 'completed'
58:           ? Ai01IntentKind.completed
59:           : Ai01IntentKind.notCompleted,
60:       status: status,
61:       executionId: executionId,
62:       evidenceRecorded: evidenceRecorded,
63:       auditId: auditId is String && auditId.isNotEmpty ? auditId : null,
64:     );
65:   }
66: }
