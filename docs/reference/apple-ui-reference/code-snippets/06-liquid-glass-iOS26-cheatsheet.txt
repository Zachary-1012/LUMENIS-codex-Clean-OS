# ============ 1. 三种材质变体 ============
# .regular  默认，工具栏/按钮/导航栏/TabBar，全自适应
# .clear    高透明，仅用于照片/地图等富媒体背景上的小型悬浮控件(需同时满足 3 个条件)
# .identity 不施加效果，用于条件开关，避免布局重算

# ============ 2. 基础 API 签名 ============
# func glassEffect<S: Shape>(
#     _ glass: Glass = .regular,
#     in shape: S = DefaultGlassEffectShape,
#     isEnabled: Bool = true
# ) -> some View
#
# struct Glass {
#     static var regular: Glass
#     static var clear: Glass
#     static var identity: Glass
#     func tint(_ color: Color) -> Glass      # 语义着色，非装饰
#     func interactive() -> Glass             # 仅 iOS：按压缩放/回弹/微光/触点扩散
# }

# ============ 3. 典型用法 ============
# Text("Hello").padding().glassEffect()
# Text("Tinted").padding().glassEffect(.regular.tint(.blue))
# Button("Tap") {}.glassEffect(.regular.interactive())
#
# # 形状
# .glassEffect(.regular, in: .capsule)
# .glassEffect(.regular, in: .circle)
# .glassEffect(.regular, in: RoundedRectangle(cornerRadius: 16))
# .glassEffect(.regular, in: .rect(cornerRadius: .containerConcentric))  # 与容器同心对齐

# ============ 4. 多元素必须包 Container（共享采样区 + 形变融合） ============

# ============ 10. 硬性设计规则 ============
# - 玻璃只用于悬浮在内容之上的"导航层"，绝不用于列表/表格/媒体等内容本身
# - 禁止 glass-on-glass 叠放；多玻璃元素必须用 GlassEffectContainer
# - 最小对比度 4.5:1；不要覆盖系统的减弱透明度/增强对比度/减弱动效设置
# - 无障碍环境值:
#   @Environment(\.accessibilityReduceTransparency) var reduceTransparency
#   @Environment(\.accessibilityReduceMotion) var reduceMotion
# - 旧设备(iPhone 11~13)注意 GPU/耗电/发热，稳态不要让玻璃持续动画
