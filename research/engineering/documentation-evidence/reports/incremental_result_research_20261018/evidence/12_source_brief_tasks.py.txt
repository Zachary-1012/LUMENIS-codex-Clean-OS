56: def _worker_alive(task):
57:     """False only when the exact claimed OS process is proven gone."""
58:     identity = task.get("parameters") or {}
59:     pid, started = identity.get("worker_pid"), identity.get("worker_started_at")
60:     if type(pid) is not int or pid <= 0 or type(started) not in {int, float}:
61:         raise RuntimeError("SOURCE_BRIEF_WORKER_IDENTITY_UNAVAILABLE")
62:     try:
63:         process = psutil.Process(pid)
64:         alive = process.is_running() and process.create_time() == started
65:         if alive and pid == os.getpid():
66:             with _lock:
67:                 worker = _workers.get(task["task_id"])
68:                 return worker is not None and worker.is_alive()
69:         return alive
70:     except psutil.NoSuchProcess:

72:     except psutil.AccessDenied as exc:
