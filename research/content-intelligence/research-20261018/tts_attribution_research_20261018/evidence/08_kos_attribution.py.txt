131: def _platform_weighted_interactions(note, platform=""):
132:     """G1: 按平台权重计算单条笔记的加权互动量。
133: 
134:     权重从 config/kos_platform_rules.json 的 platform_weights 读取。
135:     未知平台回退 xiaohongshu 权重；配置缺失时回退等权（已在 load_attribution_config 中 WARNING）。
136:     """
137:     cfg = load_attribution_config()
138:     plat = _normalize_platform(platform or note.get("source_platform", ""))
139:     pw = cfg["platform_weights"].get(
140:         plat, cfg["platform_weights"].get("xiaohongshu",
141:         {"likes": 1.0, "comments": 1.0, "collects": 1.0, "shares": 1.0}))
142:     values = {key: engagement_count(note.get(key))
143:               for key in ("likes", "comments", "collects", "shares")}
144:     if any(value is None for value in values.values()):
145:         return None
146:     return sum(float(values[key]) * pw[key] for key in values)

175: def calculate_multi_touch_attribution(note_ids: list, brand_name: str, *, scope=None) -> dict:
176:     """Calculate attribution only from notes verified in the exact selected BrandFocus."""
177:     from reprocess import _canonical_source_ids, _normalize_scope
178: 
179:     try:
180:         exact = _normalize_scope(scope, allow_global=False)
181:     except ValueError as exc:
182:         return {"ok": False, "status": "blocked", "code": str(exc)}
183:     if exact["domain_id"] != "zbmp":
184:         return {"ok": False, "status": "invalid_request", "code": "BMP_SCOPE_REQUIRED"}
185:     if not isinstance(brand_name, str) or not brand_name.strip():
186:         return {"ok": False, "status": "invalid_request", "code": "BRAND_SCOPE_REQUIRED"}
187:     if not isinstance(note_ids, list) or not note_ids or len(note_ids) > 100:
188:         return {"ok": False, "status": "invalid_request", "code": "NOTE_IDS_REQUIRED"}
189:     normalized_ids = []
190:     for note_id in note_ids:
191:         if not isinstance(note_id, str) or not note_id or note_id != note_id.strip():
192:             return {"ok": False, "status": "invalid_request", "code": "NOTE_IDS_INVALID"}
193:         if note_id not in normalized_ids:
194:             normalized_ids.append(note_id)
195: 
196:     cfg = load_attribution_config()
197:     aw = cfg["attribution_weights"]
198:     conn = _conn()
199:     try:
200:         admitted_ids = _canonical_source_ids(
201:             exact, source_ids=normalized_ids, limit=len(normalized_ids), connection=conn,
202:         )
203:         if set(admitted_ids) != set(normalized_ids):
204:             return {"ok": False, "status": "denied", "code": "CANONICAL_NOTE_SCOPE_MISMATCH"}
205: 
206:         placeholders = ",".join("?" for _ in normalized_ids)
207:         notes = conn.execute(
208:             "SELECT note_id, author_id, author_nickname, likes, comments, collects, shares, "
209:             "publish_datetime, source_platform FROM notes WHERE note_id IN (" + placeholders + ") "
210:             "AND product_id=? AND domain_id=? AND scope_status='VERIFIED' AND brand_focus_id=? "
211:             "AND COALESCE(subject_id,'')='' AND NULLIF(TRIM(scope_evidence_ref),'') IS NOT NULL "
212:             "ORDER BY publish_datetime ASC",
213:             tuple(normalized_ids) + (exact["product_id"], exact["domain_id"], exact["brand_focus_id"]),
214:         ).fetchall()
215:         if len(notes) != len(normalized_ids):
216:             return {"ok": False, "status": "denied", "code": "CANONICAL_NOTE_SCOPE_CHANGED"}
217: 
218:         missing_authors = sum(
219:             not isinstance(note["author_id"], str) or not note["author_id"].strip()
220:             for note in notes
221:         )
222:         if missing_authors:
223:             return {
224:                 "ok": False, "status": "incomplete", "code": "AUTHOR_ID_UNAVAILABLE",
225:                 "total_notes": len(notes),
226:                 "identified_author_notes": len(notes) - missing_authors,
227:                 "unattributed_note_count": missing_authors,
228:                 "attribution": [],
229:             }
230: 
231:         interactions_by_note = [
232:             _platform_weighted_interactions(dict(note)) for note in notes
233:         ]
234:         incomplete_engagement = sum(value is None for value in interactions_by_note)
235:         if incomplete_engagement:
236:             return {
237:                 "ok": False, "status": "incomplete", "code": "ENGAGEMENT_DATA_INCOMPLETE",
238:                 "total_notes": len(notes),
239:                 "engagement_complete_note_count": len(notes) - incomplete_engagement,
240:                 "engagement_incomplete_note_count": incomplete_engagement,
241:                 "attribution": [],
242:             }
243:         kos_contrib = {}
244:         for index, note in enumerate(notes):
245:             author = note["author_id"].strip()
246:             if author not in kos_contrib:
247:                 kos_contrib[author] = {
248:                     "author_nickname": note["author_nickname"], "notes": 0,
249:                     "interactions": 0, "weighted_score": 0,
250:                 }
251:             interactions = interactions_by_note[index]
252:             time_weight = float(aw["time_weight_coeff"]) + (
253:                 index / max(1, len(notes) - 1)
254:             ) * float(aw["time_weight_range"])
255:             kos_contrib[author]["notes"] += 1
256:             kos_contrib[author]["interactions"] += interactions
257:             kos_contrib[author]["weighted_score"] += interactions * time_weight
258: 
259:         total_weighted = math.fsum(item["weighted_score"] for item in kos_contrib.values())
260:         if not math.isfinite(total_weighted) or total_weighted <= 0:
261:             return {
262:                 "ok": False, "status": "unavailable",
263:                 "code": "ZERO_ENGAGEMENT_DENOMINATOR" if total_weighted == 0
264:                 else "ENGAGEMENT_DENOMINATOR_UNAVAILABLE",
265:                 "total_notes": len(notes), "total_kos": len(kos_contrib),
266:                 "total_weighted_interactions": total_weighted if math.isfinite(total_weighted) else None,
267:                 "attribution": [],
268:             }
269:         result = [
270:             {
271:                 "author_id": author,
272:                 "author_nickname": values["author_nickname"],
273:                 "note_count": values["notes"],
274:                 "interactions": round(values["interactions"], 1),
275:                 "attribution_pct": round(values["weighted_score"] / total_weighted * 100, 1),
276:             }
277:             for author, values in sorted(
278:                 kos_contrib.items(), key=lambda item: item[1]["weighted_score"], reverse=True,
279:             )
280:         ]
281:         return {
282:             "ok": True,
283:             "status": "COMPLETED",
284:             "brand_name": brand_name,
285:             "brand_focus_id": exact["brand_focus_id"],
286:             "total_notes": len(notes),
287:             "total_kos": len(kos_contrib),
288:             "total_weighted_interactions": total_weighted,
289:             "config_status": cfg["config_status"],
290:             "attribution": result,
291:             "method": "time_decay_multi_touch (canonical BrandFocus verified; platform weighted)",
292:         }
293:     finally:
294:         conn.close()
