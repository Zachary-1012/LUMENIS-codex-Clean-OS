50:   Future<void> start(ReadAloudRequest request) async {
51:     final generation = ++_generation;
52:     _poll?.cancel();
53:     _poll = null;
54:     _startupTimer?.cancel();
55:     _startupTimer = null;
56:     final normalizedText = request.text.trim();
57:     if (normalizedText.isEmpty) {
58:       _request = request;
59:       final stopped = await _stopPlatform();
60:       if (!_isCurrent(generation)) return;
61:       _setState(
62:         ReadAloudState(
63:           status: ReadAloudStatus.failed,
64:           errorCode: stopped
65:               ? ReadAloudErrorCode.textEmpty
66:               : ReadAloudErrorCode.playbackFailed,
67:         ),
68:       );
69:       return;
70:     }
71:     _request = ReadAloudRequest(
72:       sessionId: request.sessionId,
73:       scopeVersion: request.scopeVersion,
74:       text: normalizedText,
75:       locale: request.locale,
76:       rate: request.rate,
77:     );
78:     final stopped = await _stopPlatform();
79:     if (!_isCurrent(generation)) return;
80:     if (!stopped) {
81:       _setState(
82:         ReadAloudState(
83:           status: ReadAloudStatus.failed,
84:           errorCode: ReadAloudErrorCode.playbackFailed,
85:           locale: request.locale,
86:           rate: request.rate,
87:         ),
88:       );
89:       return;
90:     }
91:     late ReadAloudAvailability availability;
92:     try {
93:       availability = await refreshAvailability();
94:     } catch (_) {
95:       if (!_isCurrent(generation)) return;
96:       _setState(
97:         ReadAloudState(
98:           status: ReadAloudStatus.failed,
99:           errorCode: ReadAloudErrorCode.playbackFailed,
100:           locale: request.locale,
101:           rate: request.rate,
102:         ),
103:       );
104:       return;
105:     }
106:     if (!_isCurrent(generation)) return;
107:     if (!availability.supported) {
108:       _setState(
109:         const ReadAloudState(
110:           status: ReadAloudStatus.unavailable,
111:           errorCode: ReadAloudErrorCode.platformUnsupported,
112:         ),
113:       );
114:       return;
115:     }
116:     if (!availability.localLocales.contains(request.locale.tag)) {
117:       _setState(
118:         ReadAloudState(
119:           status: ReadAloudStatus.unavailable,
120:           errorCode: ReadAloudErrorCode.voiceUnavailable,
121:           locale: request.locale,
122:           rate: request.rate,
123:         ),
124:       );
125:       return;
126:     }
127:     late ReadAloudStartResult result;
128:     try {
129:       result = await _withBackendLock(() {
130:         if (!_isCurrent(generation)) {
131:           return Future.value(
132:             const ReadAloudStartResult(ReadAloudStatus.stopped),
133:           );
134:         }
135:         return _platform.speak(_request!);
136:       });
137:     } catch (_) {
138:       result = const ReadAloudStartResult(
139:         ReadAloudStatus.failed,
140:         ReadAloudErrorCode.playbackFailed,
141:       );
142:     }
143:     if (!_isCurrent(generation)) return;
144:     final requestAccepted =
145:         result.status == ReadAloudStatus.starting ||
146:         result.status == ReadAloudStatus.playing;
147:     _setState(
148:       ReadAloudState(
149:         status: requestAccepted ? ReadAloudStatus.starting : result.status,
150:         errorCode: result.errorCode,
151:         locale: request.locale,
152:         rate: request.rate,
153:       ),
154:     );
155:     if (requestAccepted) {
156:       _startupTimer = Timer(startupTimeout, () {
157:         unawaited(_failUnstartedPlayback(generation));
158:       });
159:       _poll = Timer.periodic(const Duration(milliseconds: 250), (_) {
160:         unawaited(_refreshPlayback(generation));
161:       });
162:     }
163:   }

166:     required String sessionId,
167:     required String scopeVersion,

169:     final current = _request;
170:     if (current?.sessionId != sessionId ||
171:         current?.scopeVersion != scopeVersion) {

184:         status: stopped ? ReadAloudStatus.stopped : ReadAloudStatus.failed,

192:   Future<void> _refreshPlayback(int generation) async {
193:     if (!_isCurrent(generation)) return;
194:     late ReadAloudPlaybackObservation observation;
195:     try {
196:       observation = await _platform.observe();
197:     } catch (_) {
198:       observation = ReadAloudPlaybackObservation.failed;
199:     }
200:     if (!_isCurrent(generation)) return;
201:     switch (observation) {
202:       case ReadAloudPlaybackObservation.starting:
203:         break;
204:       case ReadAloudPlaybackObservation.playing:
205:         _startupTimer?.cancel();
206:         _startupTimer = null;
207:         if (_state.status != ReadAloudStatus.playing) {
208:           _setState(_stateWith(ReadAloudStatus.playing));
209:         }
210:         break;
211:       case ReadAloudPlaybackObservation.paused:
212:         _startupTimer?.cancel();
213:         _startupTimer = null;
214:         _setState(_stateWith(ReadAloudStatus.paused));
215:         break;
216:       case ReadAloudPlaybackObservation.completed:
217:         _poll?.cancel();
218:         _poll = null;
219:         _startupTimer?.cancel();
220:         _startupTimer = null;
221:         _setState(_stateWith(ReadAloudStatus.completed));
222:         break;
223:       case ReadAloudPlaybackObservation.failed:
224:         _poll?.cancel();
225:         _poll = null;
226:         _startupTimer?.cancel();
227:         _startupTimer = null;
228:         _setState(
229:           ReadAloudState(
230:             status: ReadAloudStatus.failed,
231:             errorCode: ReadAloudErrorCode.playbackFailed,
232:             locale: _state.locale,
233:             rate: _state.rate,
234:           ),
235:         );
236:         break;
237:     }
238:   }

240:   Future<void> _failUnstartedPlayback(int generation) async {
241:     if (!_isCurrent(generation) || _state.status != ReadAloudStatus.starting) {
242:       return;
243:     }
244:     final timeoutGeneration = ++_generation;
245:     _poll?.cancel();
246:     _poll = null;
247:     _startupTimer?.cancel();
248:     _startupTimer = null;
249:     await _stopPlatform();
250:     if (!_isCurrent(timeoutGeneration)) return;
251:     _setState(
252:       ReadAloudState(
253:         status: ReadAloudStatus.failed,
254:         errorCode: ReadAloudErrorCode.playbackFailed,
255:         locale: _state.locale,
256:         rate: _state.rate,
257:       ),
258:     );
259:   }

266:   Future<T> _withBackendLock<T>(Future<T> Function() action) async {
267:     final previous = _backendTail;
268:     final done = Completer<void>();
269:     _backendTail = done.future;
270:     try {
271:       await previous;
272:       return await action();
273:     } finally {
274:       done.complete();
275:     }
276:   }
