ARK_URL = "https://ark.cn-beijing.volces.com/api/v3/chat/completions"


def public_opinion_analyze(content: str):
    prompt = """
你是专业舆情分析助手，对输入文本做分析，严格返回JSON，不要额外解释。
输出字段：
- summary:简短摘要
- sentiment: positive/neutral/negative
- risk_level: 0无风险 1低风险 2中风险 3高风险
- tags:数组，提取关键词标签
"""

    headers = {
        "Authorization": f"Bearer {ARK_API_KEY}",
        "Content-Type": "application/json"
    }

    payload = {
        "model": ARK_ENDPOINT_ID,
        "temperature": 0.1,
        "max_tokens": 512,
        "messages": [
            {"role": "system", "content": prompt},
            {"role": "user", "content": content}
        ]
    }

    resp = requests.post(ARK_URL, json=payload, headers=headers, timeout=30)
    resp.raise_for_status()
    data = resp.json()
    raw = data["choices"][0]["message"]["content"]
    import json
    return json.loads(raw)


from flask import Flask, request, jsonify
app = Flask(__name__)

@app.route("/api/opinion/analyze", methods=["POST"])
def api_analyze():
    req = request.get_json()
    text = req.get("text", "")
    if not text:
        return jsonify({"code":400,"msg":"text不能为空"}),400
    try:
        res = public_opinion_analyze(text)
        return jsonify({"code":0,"data":res})
    except Exception as e:
        return jsonify({"code":500,"msg":str(e)}),500

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=8100, debug=False)
